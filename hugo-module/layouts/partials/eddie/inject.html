{{- $cfg := .Site.Params.eddie | default dict -}} {{- $enabled := $cfg.enabled | default true -}} {{- if $enabled -}} {{- $indexPath := $cfg.indexUrl | default "/eddie/index.ed" -}} {{- $position := $cfg.position | default "bottom-right" -}} {{- $theme := $cfg.theme | default "auto" -}} {{- $offsetY := $cfg.offsetY | default 0 -}} {{- $offsetX := $cfg.offsetX | default 0 -}} {{- $qaMode := $cfg.qaMode | default "auto" -}} {{- $qaSubject := $cfg.qaSubject | default "" -}} {{- $topK := $cfg.topK | default 8 -}} {{- $answerTopK := $cfg.answerTopK | default 5 -}} {{- $agentMode := $cfg.agentMode | default "auto" -}} {{- $agentModel := $cfg.agentModel | default "auto" -}} {{- $denseRuntime := $cfg.denseRuntime | default "auto" -}} {{- $consentText := $cfg.consentText | default "" -}} {{- $persist := $cfg.persist | default "auto" -}} {{- $warm := $cfg.warm | default "auto" -}} {{/* relURL/absURL only prepend the site's baseURL path prefix (e.g. "/blog/" for a site served under https://example.com/blog/) to a *relative* path. A leading "/" makes Hugo treat the value as already root-relative and leaves it untouched -- which silently drops the subpath. Strip a leading "/" before calling relURL so "/eddie/index.ed" and "eddie/index.ed" behave the same; an absolute URL with a scheme has no leading "/" to strip and passes through relURL unchanged either way. */}} {{- $indexRelPath := strings.TrimPrefix "/" $indexPath -}} {{- $indexUrl := $indexRelPath | relURL -}} {{/* Cache-bust the index URL with a `?v=` derived from its content so a rebuilt index is fetched immediately instead of serving a stale cached copy. When the index lives under assets/ (Hugo Pipes can read it via resources.Get), the value is a content hash and only changes when the index actually changes. When it lives under static/ (the common case; Hugo templates can't read arbitrary static/ files), this falls back to the build timestamp, which busts the cache on every build instead of only on content changes -- acceptable since static/eddie/index.ed is typically rebuilt only when the site itself is rebuilt. This value reaches only the index, its index..ed sidecars and any site-bundled model files: things one `eddie index` run writes together. The runtime assets (eddie-widget.js, the workers, the wasm and its glue, the service workers) carry their own `?v=`, a hash of the files that built them, stamped into every bundle by widget/build.sh. Sharing this one with them re-downloaded the whole engine on every deploy. */}} {{- $cacheBust := now.Unix -}} {{- with resources.Get $indexRelPath -}} {{- $cacheBust = substr (crypto.SHA256 .Content) 0 12 -}} {{- end -}} {{- $indexUrl = printf "%s?v=%v" $indexUrl $cacheBust -}} {{/* loader = "boot" (default) puts the 2 KB eddie-boot.js on the page: it draws the trigger and the Ctrl/Cmd+K shortcut and fetches eddie-widget.js on the first interaction (or after load for a visitor who used the search before). loader = "full" loads eddie-widget.js on every page view. */}} {{- $loader := $cfg.loader | default "boot" -}} {{- $script := "eddie/eddie-boot.js" -}} {{- if eq $loader "full" -}}{{- $script = "eddie/eddie-widget.js" -}}{{- end -}} {{- end -}}
